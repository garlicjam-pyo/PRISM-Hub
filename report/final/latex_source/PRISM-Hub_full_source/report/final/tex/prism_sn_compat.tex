% The original report is a chapter-based document. For a faithful first
% transfer into the article class, reinterpret each chapter as a section,
% each report section as a subsection, and so on, without editing sources.
\let\PrismArticleSection\section
\let\PrismArticleSubsection\subsection
\let\PrismArticleSubsubsection\subsubsection
\let\PrismArticleParagraph\paragraph
\providecommand{\chapter}{}
\renewcommand{\chapter}{\PrismArticleSection}
\renewcommand{\section}{\PrismArticleSubsection}
\renewcommand{\subsection}{\PrismArticleSubsubsection}
\renewcommand{\subsubsection}{\PrismArticleParagraph}

% Let the official class control figure and table numbering and captions.
% The existing chapter-to-section mapping stays for body compatibility.
\makeatletter

\renewcommand{\thechapter}{\thesection}
\makeatother
